# f_get_dcr_no

## FUNCTION public.f_get_dcr_no(p_appr_id numeric)

- 반환: `character varying`
- 줄 수: 45

```sql
CREATE OR REPLACE FUNCTION public.f_get_dcr_no(p_appr_id numeric)
 RETURNS character varying
 LANGUAGE plpgsql
AS $function$
-----------------------------------------------------------------------------------------
DECLARE
    v_APPR      apr01_appr%ROWTYPE;
    v_DCR_NO    VARCHAR;
-----------------------------------------------------------------------------------------
BEGIN
    -------------------------------------------------------------------------------------
    SELECT *
      INTO v_APPR
      FROM apr01_appr
     WHERE apr01_appr_id = p_appr_id;

    IF v_APPR.apr01_appr_id IS NULL THEN
        RETURN NULL;
    END IF;

    /*-----------------------------------------------------------------------------------
       대외문서번호 리턴
     ------------------------------------------------------------------------------------
     - 연도별(기본) : -2024-10001
     - 연도-부서별  : -2024-운용10001
    -------------------------------------------------------------------------------------
     - 예외처리
       1. 202409021859096 피아이엠자산운용 예외처리 : [ 운용 24-1]
    ------------------------------------------------------------------------------------*/
    IF v_APPR.apr01_company_id = 202409021859096 THEN
        ---------------------------------------------------------------------------------
        v_DCR_NO := CONCAT(' ', v_APPR.apr01_dcr_id_word, ' ', TO_CHAR(f_today(), 'yy'), '-', REPLACE(v_APPR.apr01_stamp_seq, v_APPR.apr01_dcr_id_word, ''));
    -------------------------------------------------------------------------------------
    ELSE
        v_DCR_NO := CONCAT('-', TO_CHAR(f_today(), 'yyyy'), '-', v_APPR.apr01_stamp_seq);
    END IF;

    -------------------------------------------------------------------------------------
    RETURN v_DCR_NO;
-----------------------------------------------------------------------------------------
EXCEPTION WHEN OTHERS THEN RETURN NULL;
END;
$function$
```
