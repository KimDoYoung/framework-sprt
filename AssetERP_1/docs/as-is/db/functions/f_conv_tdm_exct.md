# f_conv_tdm_exct

## FUNCTION public.f_conv_tdm_exct(p_company_id numeric, p_role_nm character varying)

- 반환: `numeric`
- 줄 수: 62

```sql
CREATE OR REPLACE FUNCTION public.f_conv_tdm_exct(p_company_id numeric, p_role_nm character varying)
 RETURNS numeric
 LANGUAGE plpgsql
AS $function$
-----------------------------------------------------------------------------------------
DECLARE
    v_EMP_NO    TEXT;
    v_PERSON_ID NUMERIC;
-----------------------------------------------------------------------------------------
BEGIN
    -------------------------------------------------------------------------------------
    -- 유리 : 202508290470920
    IF p_company_id = 202508290470920 THEN
        ---------------------------------------------------------------------------------
        -- 대표이사                     => 002 조우*
        -- CIO                          => 003 한진*
        -- CMO                          => 004 오경*
        -- 경영지원본부장               => 044 전용*
        -- 준법감시인 및 위험관리책임자 => 052 장세*
        -- 감사                         => 052 장세*
        ---------------------------------------------------------------------------------
        v_EMP_NO := CASE p_role_nm WHEN '대표이사'                     THEN '002'
                                   WHEN 'CIO'                          THEN '003'
                                   WHEN 'CMO'                          THEN '004'
                                   WHEN '경영지원본부장'               THEN '044'
                                   WHEN '준법감시인 및 위험관리책임자' THEN '052'
                                   WHEN '감사'                         THEN '052'
                                   ELSE NULL END;
    -------------------------------------------------------------------------------------
    -- 아이트러스트 : 202308029131917
    ELSIF p_company_id = 202308029131917 THEN
        ---------------------------------------------------------------------------------
        -- 이사회 의장      => S001 황영기
        -- 감사             => S003 박삼철
        -- 대표이사         => 1001 김철배
        -- 준법감시인       => 1003 송한진
        -- 운용대표         => 1007 김성수
        -- 경영관리본부장   => 1001 김철배
        ---------------------------------------------------------------------------------
        v_EMP_NO := CASE REPLACE(p_role_nm,' ','') WHEN '이사회의장'        THEN 'S001'
                                                   WHEN '감사'              THEN 'S003'
                                                   WHEN '대표이사'          THEN '1001'
                                                   WHEN '준법감시인'        THEN '1003'
                                                   WHEN '운용대표'          THEN '1007'
                                                   WHEN '경영관리본부장'    THEN '1001'
                                                   ELSE NULL END;

    END IF;
    -------------------------------------------------------------------------------------
    SELECT emp01_person_id
      INTO v_PERSON_ID
      FROM emp01_person
     WHERE emp01_company_id = p_company_id
       AND emp01_emp_no     = v_EMP_NO;
    -------------------------------------------------------------------------------------
    RETURN v_PERSON_ID;
-----------------------------------------------------------------------------------------
EXCEPTION WHEN OTHERS THEN RETURN NULL;
END;
$function$
```
